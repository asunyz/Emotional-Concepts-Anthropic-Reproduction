def check_similarity(drawings):
    """
    Compares a list of current drawings against a database of past sketches.
    Returns True if a match exceeds 80% similarity.
    """
    db = load_database("childhood_art")
    matches = []
    for drawing in drawings:
        score = compare_images(drawing, db)
        if score > 0.8:
            matches.append(drawing.id)
    return len(matches) > 0
